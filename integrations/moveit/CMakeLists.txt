cmake_minimum_required(VERSION 3.16)
project(piper_moveit_offline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(urdf REQUIRED)
find_package(srdfdom REQUIRED)
find_package(yaml-cpp REQUIRED)
add_executable(ik_benchmark ik_benchmark.cpp)
ament_target_dependencies(ik_benchmark rclcpp moveit_core moveit_msgs pluginlib urdf srdfdom)
target_link_libraries(ik_benchmark yaml-cpp)
add_executable(joint_plan joint_plan.cpp)
ament_target_dependencies(joint_plan rclcpp moveit_core moveit_msgs pluginlib urdf srdfdom)
target_link_libraries(joint_plan yaml-cpp)
add_executable(ik_server ik_server.cpp)
ament_target_dependencies(ik_server rclcpp moveit_core moveit_msgs pluginlib urdf srdfdom)
target_link_libraries(ik_server yaml-cpp)
